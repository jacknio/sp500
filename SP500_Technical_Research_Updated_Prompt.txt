def select_quarter_holdings(raw_holdings, analysis_date):
    h = raw_holdings.copy()

    h["report_date"] = pd.to_datetime(h["report_date"])
    analysis_date = pd.Timestamp(analysis_date)

    # Keep holdings whose report date is in the current calendar quarter
    h = h.loc[
        h["report_date"].dt.to_period("Q")
        == analysis_date.to_period("Q")
    ].copy()

    # For each fund, find its latest report date within this quarter
    latest_date = h.groupby(
        "factset_fund_id"
    )["report_date"].transform("max")

    # Keep ALL holdings belonging to that fund's latest report date
    h = h.loc[h["report_date"].eq(latest_date)].copy()

    return h
