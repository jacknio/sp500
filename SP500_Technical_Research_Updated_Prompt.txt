def select_quarter_holdings(raw_holdings, analysis_date):
    h = raw_holdings.copy()

    h["report_date"] = pd.to_datetime(
        h["report_date"], errors="coerce"
    ).dt.normalize()

    # Only use holdings as of the analysis date
    h = h.loc[
        h["report_date"].notna()
        & h["report_date"].le(pd.Timestamp(analysis_date).normalize())
    ].copy()

    # Keep each fund's latest report date
    latest = h.groupby("factset_fund_id")["report_date"].transform("max")

    return h.loc[h["report_date"].eq(latest)].copy()


def add_holdings_sum(funds, raw_holdings, analysis_date):
    n = raw_holdings.copy()

    n["report_date"] = pd.to_datetime(
        n["report_date"], errors="coerce"
    ).dt.normalize()

    n["reported_mv"] = pd.to_numeric(
        n["reported_mv"], errors="coerce"
    )

    # Only use data available as of analysis date
    n = n.loc[
        n["report_date"].notna()
        & n["report_date"].le(pd.Timestamp(analysis_date).normalize())
    ].copy()

    # Latest report date for each fund
    latest = n.groupby("factset_fund_id")["report_date"].transform("max")
    n = n.loc[n["report_date"].eq(latest)].copy()

    # Sum ALL holdings, including unmapped securities
    aum = n.groupby("factset_fund_id", as_index=False).agg(
        fund_aum_from_holdings=(
            "reported_mv", lambda s: s.sum(min_count=1)
        ),
        aum_holdings_report_date=("report_date", "max"),
        aum_missing_mv_rows=("reported_mv", lambda s: s.isna().sum()),
    )

    # Each main loop contains one analysis date
    return funds.merge(
        aum,
        on="factset_fund_id",
        how="left",
        validate="many_to_one"
    )
