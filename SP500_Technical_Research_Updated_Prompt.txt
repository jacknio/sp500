def add_holdings_sum(funds, raw_holdings, analysis_date):
    n = raw_holdings.copy()

    n["report_date"] = pd.to_datetime(n["report_date"])
    n["reported_mv"] = pd.to_numeric(
        n["reported_mv"], errors="coerce"
    )

    # Sum all holdings for each fund and report date
    aum = (
        n.groupby(
            ["factset_fund_id", "report_date"],
            as_index=False
        )["reported_mv"]
        .sum(min_count=1)
        .rename(columns={
            "reported_mv": "fund_aum_from_holdings"
        })
    )

    # Match each fund to its corresponding report date
    funds = funds.copy()
    funds["fund_reference_date"] = pd.to_datetime(
        funds["fund_reference_date"]
    )

    aum = aum.rename(columns={
        "report_date": "fund_reference_date"
    })

    return funds.merge(
        aum,
        on=["factset_fund_id", "fund_reference_date"],
        how="left",
        validate="many_to_one"
    )
